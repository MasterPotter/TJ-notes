\begin{enumerate}
    \item \textbf{Problem 5.9.12:} Show that:
    \[\int_{0}^{\infty} \frac{(\ln z)^2}{1+z^2}dz = 4\left( 1 - \frac 1{3^3} + \frac1{5^3} - \frac1{7^3}+\cdots\right)\]
    \\ \\ 
    \[\int_{0}^{\infty} \frac{(\ln z)^2}{1+z^2}dz = \int_{0}^{1} \frac{(\ln z)^2}{1+z^2}dz + \int_{1}^{\infty} \frac{(\ln z)^2}{1+z^2}dz = 2 \int_{0}^{1} \frac{(\ln x)^2}{1+x^2}dx = 2 \sum_{n=0}^{\infty} (-1)^nx^{2n}dx \]
    \[= 2\sum_{n=0}^{\infty} (-1)^n\int_{0}^{1}x^{2n}\ln^2 x dx = 4\sum_{n=0}^{\infty}\frac{(-1)^n}{(2n+1)^3} = 4\left( 1 - \frac 1{3^3} + \frac1{5^3} - \frac1{7^3}+\cdots\right) \hspace{1mm} \checkmark\]
    \item \textbf{Problem 8.4.6:} Show, by means of the beta function, that:
    \item \[\int_{t}^{z}\frac{dx}{(z-x)^{1-\alpha}(x-t)^{\alpha}} = \frac{\pi}{\sin \pi \alpha}, 0 < \alpha < 1\]
    \\ \\ 
    \[x-t = u(z-t), z-x = (z-t)(1-u), \int_{0}^{1}\frac{(z-t)du}{((z-t)(1-u))^{1-\alpha}(u(z-t))^\alpha}= \int_{0}^{1}u^{-\alpha}(1-u)^{\alpha - 1}\]
    \[= \Gamma(1-\alpha)\Gamma(\alpha) = \boxed{\frac{\pi}{\sin \pi \alpha}} \hspace{1mm} \checkmark\]
    \item \textbf{Problem 8.4.10:} Evaluate $\int_{0}^{1} (1-x^4)^{-\frac{1}{2}}dx$ as a beta function:
    \\ \\ 
    \[t = x^4, \int_{0}^{1}(1-t)^{-\frac 1 2} \cdot \frac 1 4 \cdot t^{-\frac 3 4}dt = \frac{1}{4}B(\frac 1 4, \frac 1 2) = \frac{1}{4} \frac{\Gamma(\frac 1 4)\Gamma(\frac 1 2)}{\Gamma(\frac 3 4)} = \boxed{\frac{[(\frac 1 4)!]^2 \cdot 4}{\sqrt{2\pi}}}\]
\end{enumerate}
