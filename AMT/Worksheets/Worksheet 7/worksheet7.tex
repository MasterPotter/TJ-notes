\begin{center}\LARGE{Advanced Mathematical Techniques \\ Problem Set 7}\end{center}
\vspace{10mm}
\begin{enumerate}

\item \[\int_{0}^{1} \frac{\ln^2(1-t^2)}{t^4}dt \xrightarrow{u=t^2} \int_0^1 \frac{\ln^2(1-u)}{u^2} \frac{1}{2u^{1/2}} du=\frac{1}{2} \int_0^1 (1-u)^\varepsilon u^{-5/2+\delta} du = \frac{\Gamma(1+\varepsilon)\Gamma(-3/2+\delta)}{2\Gamma(-1/2+\varepsilon+\delta)}\]
\[ = \frac{\Gamma(1+\varepsilon)}{2} \cdot \frac{\Gamma(-1/2+\delta)}{(-3/2+\delta)(-1/2+\delta)\Gamma(1/2+\varepsilon+\delta)} \xrightarrow{\delta = 0} \frac{\Gamma(1+\varepsilon) \cdot \frac{4}{3}\sqrt{\pi}}{2 \cdot \frac{3}{4} \sqrt{\pi}\Gamma(1/2+\varepsilon)}\]
\[= \frac{e^{(2\epsilon+1)\ln 2} \Gamma(1+\varepsilon)\Gamma(1+\varepsilon)}{3\Gamma(1+2\varepsilon)} = \frac{1}{3}(\varepsilon - 1/2) \exp\{ \ln 2 + 2\varepsilon\ln 2 - \gamma(2\epsilon-2\varepsilon) + \sum_{k=2}^\infty \frac{(-1)^k\zeta(k)}{k} (2-2^k)\varepsilon^k \} \]
We only need the $\varepsilon^2$ coefficient, so we consider only
\[ \frac{4}{3} \left( 2\ln 2 - \frac{1}{2} \left( 2\ln^2 2 + \frac{\zeta(2)}{2} (-1/2) \right) \right) = \boxed{\frac{\pi^2}{9} + \frac{8}{3}\ln 2 - \frac{4}{3}\ln^2 2} \]
\item $\displaystyle \int_{0}^{1}\frac{\ln^2(1-t)}{t^3\sqrt t}$ \fbox{diverges}, since at 0, $\ln(1-t) \approx -t$, so the integral simplifies to $\displaystyle \int_{0}^{1} \frac{-1}{t\sqrt t}$, which diverges at 0.
\item \[\int_{0}^{1}\frac{\ln^2(1-t^2)\ln t}{t^3}dt \xrightarrow{u = t^2}\frac 1 4\int_0^1 \frac{\ln^2(1-u) \ln u}{2u^{2}} du \rightarrow \int_{0}^{1} u^{-2+\delta}(1-u)^\varepsilon = \frac{\Gamma(1+\varepsilon)\Gamma(1+\delta)(\varepsilon + \delta)}{\Gamma(1+\varepsilon+\delta)(1+\delta)(\delta)}\]
\[ = \frac{(\varepsilon+\delta)}{\delta(1+\delta)} \exp\{ -\gamma(1-\delta+\varepsilon-\delta) + \sum_{k=2}^\infty \frac{(-1)^k\zeta(k)}{k} [\varepsilon^k + \delta^k - (\varepsilon+\delta)^k] \} = \frac{1}{\delta} \cdot (\varepsilon+\delta) \cdot (1+\delta+\delta^2+...) \cdot \exp\{previous\} \]
We only need to consider the $\varepsilon\delta$ coefficient, so
\[-\frac 1 2(\frac{\zeta(2)}{2})(-\delta) + \frac{\zeta(3)}{3} (-3) = \boxed{\frac{\pi^2}{12} - \zeta(3)}\]
\item
\[\int_{0}^{\infty} \frac{1-e^{-3x}}{x^{\frac 3 2}} \rightarrow \int_0^\infty x^{-3/2+\delta} (1-e^{-3x}) dx = \int_0^\infty x^{-3/2+\delta} dx - \int_0^\infty x^{-3/2+\delta} e^{-3x} dx = [x^{-1/2+\delta}]_0^\infty - \frac{\Gamma(-1/2+\delta)}{3^{-1/2+\delta}}\]
The first term is useless, so let's ignore it.
\[= -3e^{-\delta\ln 3} 2\cdot 2^{-2\delta} \sqrt{\pi}\frac{\Gamma(1+2\delta)}{(1-2\delta)\Gamma(1+\delta)} = -2\sqrt{3\pi} (1+2\delta+4\delta^2+...) \exp\{ -\gamma\delta - \delta\ln 3 - 2\delta\ln 2 + \sum_{k=2}^\infty \frac{(-1)^k\zeta(k)}{k}(2^k-1)\delta^k \}\]

Since the $\delta^0$ coefficient is $ 2\sqrt{3\pi}$, the integral is \boxed{2\sqrt{3\pi}}.

\item \[\int_{0}^{\infty}\frac x {e^{3x}-1} = \int_0^\infty \frac{x e^{-3x}}{1-e^{-3x}} dx \xrightarrow{u=3x} \frac{1}{9}\int_0^\infty \frac{ue^{-u}}{1-e^{-u}} du = \frac{1}{9}\sum_{k=0}^\infty \int_0^\infty ue^{-u}e^{-uk} du\]
\[=\frac{1}{9} \sum_{k=0}^\infty \int_0^\infty u e^{-(k+1)u} du = \frac{1}{9} \sum_{k=0}^\infty \frac{\Gamma(2)}{(k+1)^2} = \frac{1}{9}\zeta(2)\Gamma(2) = \boxed{\frac{\pi^2}{54}}\]

\item 
\[
\int_0^\infty \frac{x^2 e^{2x}}{(e^{2x}-1)^2}\,dx 
= -\frac{\partial}{\partial a}\bigg|_{a=2}\int_0^\infty \frac{x}{e^{ax}-1}\,dx 
= -\frac{\partial}{\partial a}\bigg|_{a=2}\frac{\Gamma(2)\zeta(2)}{a^2} 
= \frac{2\zeta(2)}{2^3} = \frac{\pi^2}{3\cdot 2^3} = \boxed{\dfrac{\pi^2}{24}}
\]

\item 
\[
\int_0^\infty \frac{x^3}{(e^{2x}-1)^2}\,dx 
= -\frac{\partial}{\partial a}\bigg|_{a=2}\int_0^\infty \frac{x^2}{e^{ax}-1}\,dx - \int_0^\infty\frac{x^3}{e^{ax}-1}\,dx
= -\frac{\partial}{\partial a}\bigg|_{a=2}\frac{\Gamma(3)\zeta(3)}{a^3} - \frac{\Gamma(4)\zeta(4)}{a^4}
= \frac{6\zeta(3)}{a^4} - \frac{6\zeta(4)}{a^4}
\]\[= \frac{6}{16}\!\left(\zeta(3) - \frac{\pi^4}{90}\right) = \boxed{\frac{3}{8}\zeta(3) - \frac{\pi^4}{240}}
\]

\item \[\int_0^1 \frac{\ln^2(1-t^2)\ln t}{t^2}dt = \int_0^1 t^{-2+\delta}(1-t)^\varepsilon\,dt = \frac{\Gamma(1+\varepsilon)\Gamma(-1+\delta)}{\Gamma(\varepsilon+\delta)}\]
which is the same Beta function integral as \#3, so the $\varepsilon\delta$ coefficient is still $\dfrac{\pi^2}{6} - 2\zeta(3)$. 
\[
\therefore \hspace{1mm}\int_0^1 \frac{\ln^2(1-t^2)\ln t}{t^2}dt = 2!\cdot 1!\cdot\left(\frac{\pi^2}{6} - 2\zeta(3)\right) = \boxed{\frac{\pi^2}{3} - 4\zeta(3)}
\]
\end{enumerate}
